# =====================================================================
# migration-execute.cmd (Phase 2)
# =====================================================================
#
# Purpose:
#   Execute the single migration subtask defined in
#   migration-progress-tracker.md and update its status.
#
# Behavior:
#   - Read migration-progress-tracker.md (one active subtask)
#   - Identify Task ID, File Path, Change Type, and Go snippet
#   - Apply code change (create or modify file)
#   - Record result (DONE / FAILED) with timestamp and notes
#   - Output concise summary and console log
#
# Inputs:
#   - migration-progress-tracker.md
#   - snippet embedded in that file
#
# Outputs:
#   - Updated migration-progress-tracker.md
#   - Log file (migration-execute.log)
# =====================================================================

config:
  tracker_file: migration-progress-tracker.md
  workspace_root: .
  go_mod: go.mod
  log_file: migration-execute.log

# ------------------------------------------------------------
# 1. Read current tracker file
# ------------------------------------------------------------
read_tracker:
  file: "${config.tracker_file}"
  parse:
    - Task ID
    - File Path
    - Change Type
    - Status
    - Go snippet block (between ```go and ```)

# ------------------------------------------------------------
# 2. Validate tracker state
# ------------------------------------------------------------
validate_state:
  if status != "TODO":
    print: "No pending subtask found or already completed. Run migration-iterate.cmd to fetch next."
    exit: true

# ------------------------------------------------------------
# 3. Prepare workspace
# ------------------------------------------------------------
prepare_workspace:
  ensure_exists: "${config.workspace_root}"
  ensure_go_mod: "${config.workspace_root}/${config.go_mod}"

# ------------------------------------------------------------
# 4. Perform code action
# ------------------------------------------------------------
perform_action:
  input: tracker_file
  workspace: "${config.workspace_root}"
  logic:
    - if change_type == "Create":
        actions:
          - create_file:
              path: ${file_path}
              content: |
                package service

                import (
                    as "github.com/aerospike/aerospike-client-go/v7"
                )

                // Auto-generated by migration-execute.cmd
                // TODO: Review and customize as needed.

                ${go_snippet}
    - if change_type == "Modify":
        actions:
          - modify_file:
              path: ${file_path}
              rules:
                - search: "redis."
                  replace: "// redis call deprecated\n// replaced with Aerospike logic\n${go_snippet}"
                - search: "import .*redis"
                  replace: "import as \"github.com/aerospike/aerospike-client-go/v7\""
          - append_comment:
              path: ${file_path}
              text: "// Migration note: updated from Redis → Aerospike"
    - if change_type == "Migrate":
        actions:
          - create_file:
              path: "scripts/migrate-redis-to-aerospike.go"
              content: |
                package main

                import (
                    as "github.com/aerospike/aerospike-client-go/v7"
                    "github.com/go-redis/redis/v8"
                    "context"
                    "fmt"
                )

                // Auto-generated data migration skeleton
                func main() {
                    fmt.Println("Starting Redis → Aerospike migration")
                    ${go_snippet}
                }
    - if change_type == "Test":
        actions:
          - create_file:
              path: ${file_path}
              content: |
                package service_test

                import (
                    "testing"
                    as "github.com/aerospike/aerospike-client-go/v7"
                )

                func TestAerospikeIntegration(t *testing.T) {
                    ${go_snippet}
                }

# ------------------------------------------------------------
# 5. Verify changes
# ------------------------------------------------------------
verify_changes:
  checks:
    - file_exists: ${file_path}
    - if change_type == "Modify":
        ensure_contains: "Aerospike"
    - if change_type == "Create":
        ensure_not_empty: ${file_path}
  on_failure:
    goto: update_tracker_failed

# ------------------------------------------------------------
# 6. Update tracker (DONE)
# ------------------------------------------------------------
update_tracker_done:
  input: tracker_file
  rewrite: |
    # migration-progress-tracker.md
    ## Tracker Summary
    | Task ID | File Path | Change Type | Status | Completed At | Notes |
    |----------|------------|-------------|---------|---------------|-------|
    | ${task_id} | ${file_path} | ${change_type} | DONE | <timestamp> | Code change applied successfully |
    ---
    ## Current Subtask
    ### ${task_id} completed
    ✅ Migration step executed successfully.
    File changed: ${file_path}
    ---
  log: "Task ${task_id} completed successfully"

# ------------------------------------------------------------
# 7. Update tracker (FAILED)
# ------------------------------------------------------------
update_tracker_failed:
  input: tracker_file
  rewrite: |
    # migration-progress-tracker.md
    ## Tracker Summary
    | Task ID | File Path | Change Type | Status | Completed At | Notes |
    |----------|------------|-------------|---------|---------------|-------|
    | ${task_id} | ${file_path} | ${change_type} | FAILED | <timestamp> | Error applying code changes |
    ---
    ## Current Subtask
    ### ${task_id} failed
    ❌ Migration step failed. Review error logs and rerun.
    ---
  log: "Task ${task_id} failed. Manual intervention required."

# ------------------------------------------------------------
# 8. Output summary
# ------------------------------------------------------------
output_summary:
  - print: "Migration executed for ${file_path}"
  - print: "Status: DONE or FAILED (see tracker for details)"
  - print: "Log file: ${config.log_file}"
  - write_console_log: "${config.log_file}"
