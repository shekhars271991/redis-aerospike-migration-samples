# =====================================================================
# migration-iterate.cmd (Phase 1)
# =====================================================================
#
# Purpose:
#   Incrementally expand the Redis → Aerospike migration plan into
#   one detailed subtask at a time and maintain up-to-date
#   data model mappings.
#
# Behavior:
#   - Read migration-strategy.md
#   - Read migration-progress-tracker.md (create if not exists)
#   - Read data-modeling-guidelines.md
#   - Select the next migration item not yet expanded
#   - Generate **one granular subtask** for it with:
#       • Description of the change
#       • Relevant Go code snippet
#       • Corresponding data-model update
#   - Mark the previous task as DONE (if completed via migration-execution)
#     or SKIPPED (if user didn’t execute it before this run)
#   - Write/refresh the migration-progress-tracker.md so it contains:
#       1. Historical summary (DONE/SKIPPED)
#       2. Only one active subtask marked TODO
#   - Wait for Phase 2 (migration-execution) to mark it complete
#     before continuing to the next subtask on next run.
#
# Notes:
#   This command does NOT modify code.
#   It prepares the next actionable migration slice.
# =====================================================================

config:
  plan_file: migration-strategy.md
  tracker_file: migration-progress-tracker.md
  data_model_file: data-modeling-guidelines.md
  snippet_library: snippet-library.yaml
  workspace_root: .
  namespace_default: quiz

# ------------------------------------------------------------
# 1. Read input files
# ------------------------------------------------------------
read_inputs:
  - read_file: "${config.plan_file}"
  - optional_read_file: "${config.tracker_file}"
  - read_file: "${config.data_model_file}"

# ------------------------------------------------------------
# 2. Parse migration plan
# ------------------------------------------------------------
parse_migration_plan:
  input: migration-strategy.md
  extract_sections:
    - "## 2. Impacted Files Summary"
    - "## 3. Migration Plan Summary"
  output: plan_tasks.json
  rules:
    - Each impacted file or migration action = one high-level task
    - Tasks labeled sequentially: TASK-001, TASK-002, ...
    - Extract:
        * File Path
        * Change Type
        * Description
    - Infer `key_type` (hash, string, list, set, sorted_set, ttl, etc.)
      from file context or description keywords (e.g., ZADD → sorted_set, HGET → hash).

# ------------------------------------------------------------
# 3. Ensure tracker file exists or initialize it
# ------------------------------------------------------------
check_tracker:
  if_missing:
    create_file: "${config.tracker_file}"
    template: |
      # migration-progress-tracker.md
      # Auto-generated by migration-iterate.cmd.
      # This file tracks the current subtask being migrated.
      ---
      ## Tracker Summary
      | Task ID | File Path | Change Type | Status | Updated At | Notes |
      |----------|------------|-------------|---------|-------------|-------|
      (empty)
      ---
      ## Current Subtask
      *(empty – will be populated when this command runs)*

# ------------------------------------------------------------
# 4. Parse existing tracker state
# ------------------------------------------------------------
parse_tracker:
  input: "${config.tracker_file}"
  output: tracker_state.json
  rules:
    - Extract current Task ID, Status, and Summary Table
    - Detect the most recent task’s status (TODO, DONE, SKIPPED)

# ------------------------------------------------------------
# 5. Determine next migration item
# ------------------------------------------------------------
select_next_task:
  plan: plan_tasks.json
  tracker: tracker_state.json
  output: current_task.json
  logic:
    - If the tracker has an unfinished subtask (Status == TODO):
        * Mark it as SKIPPED with timestamp
    - Then select the next plan task not present in tracker summary
    - If all plan tasks already tracked → print "All migration tasks expanded" and exit

# ------------------------------------------------------------
# 6. Generate next granular subtask
# ------------------------------------------------------------
generate_next_subtask:
  input: current_task.json
  output: next_subtask.json
  rules:
    - Only one subtask is generated per run.
    - Example mappings:
        * Create → “Scaffold <file_path> and add Aerospike DAO skeleton”
        * Modify → “Replace Redis call in <file> with Aerospike equivalent”
        * Migrate → “Implement Redis → Aerospike data migration script for <key_type>”
        * Test → “Create Aerospike integration test for <component>”
    - Attach inferred `key_type` to guide snippet selection.

# ------------------------------------------------------------
# 7. Attach Go snippet (from snippet library)
# ------------------------------------------------------------
attach_snippet:
  input: next_subtask.json
  library: "${config.snippet_library}"
  match_by: key_type
  output: next_subtask_with_snippet.json
  behavior:
    - Embed fenced ```go code``` snippet based on key_type
    - Add note: "Review and customize namespace/bins before execution."

# ------------------------------------------------------------
# 8. Update data-modeling-guidelines.md
# ------------------------------------------------------------
update_data_model_mapping:
  input: next_subtask_with_snippet.json
  data_model_file: "${config.data_model_file}"
  section_title: "## 2. Redis to Aerospike Key Mappings (Auto-Generated)"
  update_rules:
    - For each subtask, extract Redis key pattern
    - Add or update Aerospike mapping entry with:
        * Redis key example
        * Aerospike namespace
        * Set name
        * Record key
        * Bin structure
        * Notes
    - Avoid duplicates across runs.

# ------------------------------------------------------------
# 9. Refresh migration-progress-tracker.md
# ------------------------------------------------------------
rewrite_tracker:
  input: next_subtask_with_snippet.json
  previous_state: tracker_state.json
  target_file: "${config.tracker_file}"
  overwrite_with: |
    # migration-progress-tracker.md
    # Tracks current migration state.
    ---
    ## Tracker Summary
    | Task ID | File Path | Change Type | Status | Updated At | Notes |
    |----------|------------|-------------|---------|-------------|-------|
    ${for_each_prev_task:
      | ${prev_task.id} | ${prev_task.file_path} | ${prev_task.change_type} | ${prev_task.status == "TODO" ? "SKIPPED" : prev_task.status} | <timestamp> | Auto-updated by migration-iterate.cmd |
    }
    | ${task.id} | ${task.file_path} | ${task.change_type} | TODO | <timestamp> | New subtask generated |
    ---
    ## Current Subtask

    ### ${task.title}
    **File:** ${task.file_path}  
    **Type:** ${task.change_type}  
    **Status:** TODO  
    **Created:** <timestamp>

    ${subtask.description}

    ${subtask.snippet}

    ---
    _Note: Run migration-execution.cmd to complete this subtask before continuing._

# ------------------------------------------------------------
# 10. Log summary
# ------------------------------------------------------------
log_summary:
  - print: "Prepared next subtask: ${task.id} (${task.file_path})"
  - print: "Marked any unfinished previous tasks as SKIPPED."
  - print: "Updated ${config.data_model_file} with latest Redis→Aerospike mapping."
  - print: "Tracker file refreshed with single active subtask."
  - write_console_log: migration-iterate.log

# =====================================================================
# Embedded Code Snippet Library (Go examples)
# =====================================================================
snippet_library:
  hash: |
    ```go
    // Redis Hash → Aerospike Record

        as.NewBin("name", name),
        as.NewBin("email", email),
    ```

  string: |
    ```go

    ```
  list: |
    ```go
    // Redis List → Aerospike Ordered List
    key, _ := as.NewKey("quiz", "notifications", userID)
    list := []interface{}{"alert1", "alert2", "alert3"}
    err := client.PutBins(nil, key, as.NewBin("alerts", list))
    ```
  set: |
    ```go
    // Redis Set → Aerospike List (unique elements)
    key, _ := as.NewKey("quiz", "roles", userID)
    roles := []interface{}{"admin", "editor"}
    err := client.PutBins(nil, key, as.NewBin("roles", roles))
    ```
  sorted_set: |
    ```go
    // Redis Sorted Set → Aerospike Ordered List of maps
    key, _ := as.NewKey("quiz", "leaderboard", "global")
    scores := []interface{}{
        map[string]interface{}{"player": "p1", "score": 120},
        map[string]interface{}{"player": "p2", "score": 80},
    }
    err := client.PutBins(nil, key, as.NewBin("scores", scores))
    ```
  ttl: |
    ```go
    // Redis TTL key → Aerospike Record with TTL
    key, _ := as.NewKey("quiz", "sessions", sessionID)
    wp := as.NewWritePolicy(0, 1800)
    err := client.PutBins(wp, key, as.NewBin("data", sessionData))
    ```
